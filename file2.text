 simultanously LED blink 

#define led1 2
#define led2 4
#define led3 7
void setup() {
  pinMode(led1,OUTPUT);
  Serial.begin(1000);
  pinMode(led2,OUTPUT);
  Serial.begin(1000);
  pinMode(led3,OUTPUT);
  Serial.begin(1000);
}
void loop(){
  digitalWrite(led1,HIGH);
   digitalWrite(led2,HIGH);
   digitalWrite(led3,HIGH);
  delay(1000);
   digitalWrite(led1,LOW);
   digitalWrite(led2,LOW);
   digitalWrite(led3,LOW);
  delay(1000);
}
