parallel LED blinks 

#define my_led1 2 
#define my_led2 4
#define my_led3 7
void setup() {
pinMode(my_led1, OUTPUT);
Serial.begin(9600);
pinMode(my_led2, OUTPUT);
Serial.begin(9600);
pinMode(my_led3, OUTPUT);
Serial.begin(9600);//to begin serial communication
}

void loop() {
digitalWrite(my_led1, HIGH);
delay(1000);
digitalWrite(my_led1,LOW);
delay(1000);
digitalWrite(my_led2, HIGH);
delay(1000);
digitalWrite(my_led2,LOW);
delay(1000);
digitalWrite(my_led3, HIGH);
delay(1000);
digitalWrite(my_led3,LOW);
delay(1000);
}
